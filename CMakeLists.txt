cmake_minimum_required(VERSION 3.24)
project(fta2-native VERSION 0.1.35 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(FTA2_FETCH_SDL "Download pinned SDL if not installed" ON)
find_package(SDL3 3.4.18 EXACT CONFIG QUIET)
if(NOT SDL3_FOUND)
 if(NOT FTA2_FETCH_SDL)
  message(FATAL_ERROR "SDL 3.4.18 is required; set SDL3_DIR or enable FTA2_FETCH_SDL")
 endif()
 include(FetchContent)
 set(SDL_SHARED ON CACHE BOOL "" FORCE)
 set(SDL_STATIC OFF CACHE BOOL "" FORCE)
 set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
 FetchContent_Declare(SDL3 URL https://github.com/libsdl-org/SDL/releases/download/release-3.4.18/SDL3-3.4.18.tar.gz URL_HASH SHA256=9c75cf16330322c217dedd2e0609f1124f1b54b8633e763467b4684d0f4334a3 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
 FetchContent_MakeAvailable(SDL3)
endif()
file(GLOB SAGA2_SOURCES CONFIGURE_DEPENDS src/saga2/*.cpp)
add_library(fta2-engine STATIC ${SAGA2_SOURCES} src/platform/runtime.cpp src/platform/resources.cpp src/platform/probe.cpp src/platform/saves.cpp src/platform/startup.cpp src/audio/midiparser.cpp src/audio/midiparser_xmidi.cpp src/audio/miles_adlib.cpp src/audio/nuked.cpp)
target_sources(fta2-engine PRIVATE src/platform/widescreen_probe.cpp src/platform/diagnostics.cpp)
target_include_directories(fta2-engine PUBLIC src)
target_link_libraries(fta2-engine PUBLIC SDL3::SDL3)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS assets/fta2.png)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/assets/fta2.png" FTA2_ICON_HEX HEX)
string(REGEX REPLACE "(..)" "0x\\1,\n" FTA2_ICON_BYTES "${FTA2_ICON_HEX}")
configure_file(cmake/app-icon.h.in generated/app-icon.h @ONLY)
target_include_directories(fta2-engine PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(MSVC)
 target_compile_options(fta2-engine PRIVATE /W3 /permissive-)
 target_compile_definitions(fta2-engine PUBLIC _CRT_SECURE_NO_WARNINGS NOMINMAX)
else()
 target_compile_options(fta2-engine PRIVATE -Wno-multichar -Wno-address-of-packed-member)
endif()
add_executable(fta2-native src/main.cpp)
if(WIN32)
 enable_language(RC)
 configure_file(cmake/app-icon.rc.in generated/app-icon.rc @ONLY)
 set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/generated/app-icon.rc" PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/fta2.ico")
 target_sources(fta2-native PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated/app-icon.rc")
endif()
target_link_libraries(fta2-native PRIVATE fta2-engine)
target_compile_definitions(fta2-native PRIVATE FTA2_VERSION="${PROJECT_VERSION}")
set_target_properties(fta2-native PROPERTIES BUILD_RPATH "$ORIGIN;$ORIGIN/../lib" INSTALL_RPATH "$ORIGIN/../lib;@loader_path/../lib")
install(TARGETS fta2-native RUNTIME DESTINATION bin)
install(FILES assets/fta2.png assets/fta2.ico DESTINATION share/fta2-native/icons)
if(UNIX AND NOT APPLE)
 install(FILES assets/fta2.png DESTINATION share/icons/hicolor/512x512/apps RENAME fta2-native.png)
 install(FILES assets/fta2-native.desktop DESTINATION share/applications)
endif()
install(FILES LICENSE third_party/SCUMMVM-COPYRIGHT DESTINATION share/fta2-native)
include(CTest)
set(CPACK_GENERATOR "ZIP")
if(BUILD_TESTING)
 add_executable(fta2-voice-test tests/voice_test.cpp)
 target_link_libraries(fta2-voice-test PRIVATE fta2-engine)
 add_test(NAME voice COMMAND fta2-voice-test)
 add_executable(fta2-runtime-test tests/runtime_test.cpp)
 target_link_libraries(fta2-runtime-test PRIVATE fta2-engine)
 add_test(NAME runtime COMMAND fta2-runtime-test)
 set_tests_properties(runtime PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
 find_package(Python3 COMPONENTS Interpreter)
 if(Python3_FOUND)
  add_executable(fta2-diagnostics-test tests/diagnostics_test.cpp)
  target_link_libraries(fta2-diagnostics-test PRIVATE fta2-engine)
  add_test(NAME diagnostics COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/diagnostics_test.py --binary $<TARGET_FILE:fta2-diagnostics-test> --emulator ${CMAKE_CROSSCOMPILING_EMULATOR})
  add_test(NAME data-tool COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/data_test.py)
  add_test(NAME complete-package COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/package_test.py)
 endif()
 add_test(NAME cli-help COMMAND fta2-native --help)
 add_test(NAME cli-version COMMAND fta2-native --version)
endif()
get_target_property(FTA2_SDL_TARGET SDL3::SDL3 ALIASED_TARGET)
if(NOT FTA2_SDL_TARGET)
 set(FTA2_SDL_TARGET SDL3::SDL3)
endif()
get_target_property(FTA2_SDL_IMPORTED ${FTA2_SDL_TARGET} IMPORTED)
if(FTA2_SDL_IMPORTED)
 install(IMPORTED_RUNTIME_ARTIFACTS ${FTA2_SDL_TARGET} RUNTIME DESTINATION bin LIBRARY DESTINATION lib FRAMEWORK DESTINATION lib)
else()
 install(TARGETS ${FTA2_SDL_TARGET} RUNTIME DESTINATION bin LIBRARY DESTINATION lib FRAMEWORK DESTINATION lib)
endif()
if(WIN32)
 add_custom_command(TARGET fta2-native POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL3::SDL3> $<TARGET_FILE_DIR:fta2-native>)
 set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION bin)
 include(InstallRequiredSystemLibraries)
endif()
install(FILES third_party/NUKED-LGPL-2.1 third_party/SDL-LICENSE.txt README.md AGENTS.md DESTINATION share/fta2-native)
install(FILES third_party/README.md DESTINATION share/fta2-native RENAME THIRD_PARTY.md)
install(DIRECTORY docs DESTINATION share/fta2-native)
install(DIRECTORY research third_party DESTINATION share/fta2-native)
set(CPACK_SOURCE_GENERATOR "ZIP")
set(CPACK_SOURCE_PACKAGE_FILE_NAME "fta2-native-${PROJECT_VERSION}-source")
set(CPACK_SOURCE_IGNORE_FILES "/[.]git/;/build[^/]*/;/[.]cache/;/dist/;/artifacts/;/original-game/;__pycache__;[.]pyc$")
set(CPACK_PACKAGE_FILE_NAME "fta2-native-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}")
include(CPack)
