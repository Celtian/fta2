cmake_minimum_required(VERSION 3.24)
project(fta2-ce VERSION 0.0.4 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(FTA2_FETCH_SDL "Download pinned SDL if not installed" ON)
find_package(SDL3 3.4.18 EXACT CONFIG QUIET)
if(NOT SDL3_FOUND)
 if(NOT FTA2_FETCH_SDL)
  message(FATAL_ERROR "SDL 3.4.18 is required; set SDL3_DIR or enable FTA2_FETCH_SDL")
 endif()
 include(FetchContent)
 set(SDL_SHARED ON CACHE BOOL "" FORCE)
 set(SDL_STATIC OFF CACHE BOOL "" FORCE)
 set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
 FetchContent_Declare(SDL3 URL https://github.com/libsdl-org/SDL/releases/download/release-3.4.18/SDL3-3.4.18.tar.gz URL_HASH SHA256=9c75cf16330322c217dedd2e0609f1124f1b54b8633e763467b4684d0f4334a3 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
 FetchContent_MakeAvailable(SDL3)
endif()
file(GLOB SAGA2_SOURCES CONFIGURE_DEPENDS src/saga2/*.cpp)
find_package(Python3 3.10 COMPONENTS Interpreter REQUIRED)
set(FTA2_LOCALIZATION_CATALOGS)
set(FTA2_LOCALIZATION_ARGS)
foreach(language cs es fr)
 foreach(kind ui game)
  set(catalog "${CMAKE_CURRENT_SOURCE_DIR}/localization/${language}-${kind}.json")
  list(APPEND FTA2_LOCALIZATION_CATALOGS "${catalog}")
  list(APPEND FTA2_LOCALIZATION_ARGS --catalog "${catalog}")
 endforeach()
endforeach()
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/localization-data.inc"
 COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tools/localization.py" compile ${FTA2_LOCALIZATION_ARGS} --output "${CMAKE_CURRENT_BINARY_DIR}/generated/localization-data.inc"
 DEPENDS ${FTA2_LOCALIZATION_CATALOGS} tools/localization.py VERBATIM)
add_library(fta2-engine STATIC ${SAGA2_SOURCES} src/platform/runtime.cpp src/platform/resources.cpp src/platform/probe.cpp src/platform/saves.cpp src/platform/startup.cpp src/audio/midiparser.cpp src/audio/midiparser_xmidi.cpp src/audio/miles_adlib.cpp src/audio/nuked.cpp)
target_sources(fta2-engine PRIVATE src/platform/widescreen_probe.cpp src/platform/diagnostics.cpp src/platform/localization.cpp "${CMAKE_CURRENT_BINARY_DIR}/generated/localization-data.inc")
target_sources(fta2-engine PRIVATE src/platform/localization_probe.cpp)
target_sources(fta2-engine PRIVATE src/platform/screenshots.cpp)
target_include_directories(fta2-engine PUBLIC src)
target_link_libraries(fta2-engine PUBLIC SDL3::SDL3)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS assets/fta2.png)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/assets/fta2.png" FTA2_ICON_HEX HEX)
string(REGEX REPLACE "(..)" "0x\\1,\n" FTA2_ICON_BYTES "${FTA2_ICON_HEX}")
configure_file(cmake/app-icon.h.in generated/app-icon.h @ONLY)
target_include_directories(fta2-engine PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(MSVC)
 target_compile_options(fta2-engine PRIVATE /W3 /permissive- /bigobj)
 target_compile_definitions(fta2-engine PUBLIC _CRT_SECURE_NO_WARNINGS NOMINMAX)
else()
 target_compile_options(fta2-engine PRIVATE -Wno-multichar -Wno-address-of-packed-member)
endif()
add_executable(fta2-ce src/main.cpp)
if(WIN32)
 enable_language(RC)
 configure_file(cmake/app-icon.rc.in generated/app-icon.rc @ONLY)
 set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/generated/app-icon.rc" PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/fta2.ico")
 target_sources(fta2-ce PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated/app-icon.rc")
 add_executable(fta2-launcher WIN32 src/platform/windows_launcher.cpp "${CMAKE_CURRENT_BINARY_DIR}/generated/app-icon.rc")
 set_target_properties(fta2-launcher PROPERTIES OUTPUT_NAME "Faery Tale Adventure II")
 target_compile_definitions(fta2-launcher PRIVATE UNICODE _UNICODE)
 if(MINGW)
  target_link_options(fta2-launcher PRIVATE -municode)
 endif()
 install(TARGETS fta2-launcher RUNTIME DESTINATION .)
endif()
target_link_libraries(fta2-ce PRIVATE fta2-engine)
target_compile_definitions(fta2-engine PUBLIC FTA2_VERSION="${PROJECT_VERSION}")
set_target_properties(fta2-ce PROPERTIES BUILD_RPATH "$ORIGIN;$ORIGIN/../lib" INSTALL_RPATH "$ORIGIN/../lib;@loader_path/../lib")
install(TARGETS fta2-ce RUNTIME DESTINATION bin)
if(UNIX AND NOT APPLE)
 install(PROGRAMS packaging/Play.sh DESTINATION . RENAME "Faery Tale Adventure II")
elseif(APPLE)
 install(PROGRAMS packaging/Play.sh DESTINATION . RENAME "Faery Tale Adventure II.command")
endif()
install(FILES packaging/START-HERE.txt DESTINATION .)
install(FILES LICENSE third_party/SCUMMVM-COPYRIGHT third_party/NUKED-LGPL-2.1
 third_party/SDL-LICENSE.txt third_party/SCUMMVM-ICONS-GPL-2.0.txt
 packaging/THIRD-PARTY.txt DESTINATION licenses)
if(MINGW)
 install(DIRECTORY third_party/windows-runtime/ DESTINATION licenses/windows-runtime)
endif()
include(CTest)
set(CPACK_GENERATOR "ZIP")
if(BUILD_TESTING)
 add_executable(fta2-screenshots-test tests/screenshots_test.cpp)
 target_link_libraries(fta2-screenshots-test PRIVATE fta2-engine)
 add_test(NAME screenshots COMMAND fta2-screenshots-test)
 set_tests_properties(screenshots PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
 add_executable(fta2-localization-test tests/localization_test.cpp)
 target_link_libraries(fta2-localization-test PRIVATE fta2-engine)
 add_test(NAME localization COMMAND fta2-localization-test)
 add_test(NAME localization-catalog COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/localization_test.py)
 add_executable(fta2-voice-test tests/voice_test.cpp)
 target_link_libraries(fta2-voice-test PRIVATE fta2-engine)
 add_test(NAME voice COMMAND fta2-voice-test)
 add_executable(fta2-runtime-test tests/runtime_test.cpp)
 target_link_libraries(fta2-runtime-test PRIVATE fta2-engine)
 add_test(NAME runtime COMMAND fta2-runtime-test)
 set_tests_properties(runtime PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
 find_package(Python3 COMPONENTS Interpreter)
 if(Python3_FOUND)
  add_executable(fta2-diagnostics-test tests/diagnostics_test.cpp)
  target_link_libraries(fta2-diagnostics-test PRIVATE fta2-engine)
  add_test(NAME diagnostics COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/diagnostics_test.py --binary $<TARGET_FILE:fta2-diagnostics-test> --application-binary $<TARGET_FILE:fta2-ce> --emulator ${CMAKE_CROSSCOMPILING_EMULATOR})
  add_test(NAME data-tool COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/data_test.py)
  add_test(NAME complete-package COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/package_test.py)
  add_test(NAME release-packages COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/release_test.py)
 add_test(NAME reference-comparison COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/reference_compare_test.py)
 endif()
 add_test(NAME cli-help COMMAND fta2-ce --help)
 add_test(NAME cli-version COMMAND fta2-ce --version)
endif()
get_target_property(FTA2_SDL_TARGET SDL3::SDL3 ALIASED_TARGET)
if(NOT FTA2_SDL_TARGET)
 set(FTA2_SDL_TARGET SDL3::SDL3)
endif()
get_target_property(FTA2_SDL_IMPORTED ${FTA2_SDL_TARGET} IMPORTED)
if(FTA2_SDL_IMPORTED)
 install(IMPORTED_RUNTIME_ARTIFACTS ${FTA2_SDL_TARGET} RUNTIME DESTINATION bin LIBRARY DESTINATION lib FRAMEWORK DESTINATION lib)
else()
 install(TARGETS ${FTA2_SDL_TARGET} RUNTIME DESTINATION bin LIBRARY DESTINATION lib FRAMEWORK DESTINATION lib)
endif()
if(WIN32)
 add_custom_command(TARGET fta2-ce POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL3::SDL3> $<TARGET_FILE_DIR:fta2-ce>)
 set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION bin)
 include(InstallRequiredSystemLibraries)
endif()
set(CPACK_SOURCE_GENERATOR "ZIP")
set(CPACK_SOURCE_PACKAGE_FILE_NAME "fta2-ce-${PROJECT_VERSION}-source")
set(CPACK_SOURCE_IGNORE_FILES "/[.]git/;/build[^/]*/;/[.]cache/;/dist/;/artifacts/;/screenshots/;/original-game/;__pycache__;[.]pyc$")
set(FTA2_PACKAGE_ARCH "${CMAKE_SYSTEM_PROCESSOR}")
if(FTA2_PACKAGE_ARCH MATCHES "^(AMD64|amd64|x86_64)$")
 set(FTA2_PACKAGE_ARCH x86_64)
endif()
if(APPLE AND CMAKE_OSX_ARCHITECTURES MATCHES "arm64" AND CMAKE_OSX_ARCHITECTURES MATCHES "x86_64")
 set(FTA2_PACKAGE_ARCH universal)
endif()
set(CPACK_PACKAGE_FILE_NAME "fta2-ce-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-${FTA2_PACKAGE_ARCH}")
include(CPack)
